// Week 3 - MongoDB queries used to answer the required questions
// Run these in the Mongo shell (mongosh) after running week3_mongo_commands.txt
// Reference "today" for these sample queries: 2026-09-23 (last 30 days = on/after 2026-08-24)

// 1. Student summary: profile plus embedded notes for a specific student (studentId 1)
db.students.findOne(
  { studentId: 1 },
  { _id: 0, studentId: 1, firstName: 1, lastName: 1, email: 1, program: 1, notes: 1 }
)

// 2. Recent support visits: all support visits for a specific student in the last 30 days (studentId 1)
db.students.aggregate([
  { $match: { studentId: 1 } },
  { $project: {
      _id: 0,
      studentId: 1,
      firstName: 1,
      lastName: 1,
      recentVisits: {
        $filter: {
          input: "$supportVisits",
          as: "visit",
          cond: { $gte: ["$$visit.date", "2026-08-24"] }
        }
      }
  }}
])

// 3. High-engagement students: students with 3 or more support visits
db.students.aggregate([
  { $project: {
      _id: 0,
      studentId: 1,
      firstName: 1,
      lastName: 1,
      visitCount: { $size: "$supportVisits" }
  }},
  { $match: { visitCount: { $gte: 3 } } }
])
